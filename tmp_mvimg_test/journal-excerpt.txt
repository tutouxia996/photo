  /* ---------- the recorder: every mark can be replayed in order ------- */
  var REC=null;
  function emit(fn){ if(REC)REC.push(fn); else fn(RECCTX); }
  var RECCTX=null;
  function Rec(){this.ops=[];}
  Rec.prototype.push=function(f){this.ops.push(f);};
  Rec.prototype.count=function(){return this.ops.length;};
  Rec.prototype.playRange=function(c,a,b){
    for(var i=a;i<b&&i<this.ops.length;i++)this.ops[i](c);
  };
  function record(fn){
    var r=new Rec(),prev=REC;REC=r;
    fn();
    REC=prev;return r;
  }

  /* ---------- marks ---------- */
  function hexToRgb(h){h=h.replace("#","");
    if(h.length===3)h=h[0]+h[0]+h[1]+h[1]+h[2]+h[2];
    return [parseInt(h.slice(0,2),16),parseInt(h.slice(2,4),16),parseInt(h.slice(4,6),16)];}
  function rgbStr(c,a){return "rgba("+(c[0]|0)+","+(c[1]|0)+","+(c[2]|0)+","+(a==null?1:a).toFixed(3)+")";}
  function mixC(a,b,t){return [a[0]+(b[0]-a[0])*t,a[1]+(b[1]-a[1])*t,a[2]+(b[2]-a[2])*t];}
  function lite(c,t){return mixC(c,[255,252,244],t);}
  function dark(c,t){return mixC(c,[22,18,26],t);}
  function rng(seed){return gRnd(seed);}
  function smooth(pts,close){
    if(pts.length<2)return"";var p=pts.slice(),d,i;
    if(close){p.unshift(pts[pts.length-1]);p.push(pts[0],pts[1]);}
    else{p.unshift(pts[0]);p.push(pts[pts.length-1]);}
    d="M"+p[1][0].toFixed(2)+" "+p[1][1].toFixed(2);
    for(i=1;i<p.length-2;i++){var a=p[i-1],b=p[i],c=p[i+1],e=p[i+2];
      d+="C"+(b[0]+(c[0]-a[0])/6).toFixed(2)+" "+(b[1]+(c[1]-a[1])/6).toFixed(2)+","+
         (c[0]-(e[0]-b[0])/6).toFixed(2)+" "+(c[1]-(e[1]-b[1])/6).toFixed(2)+","+
         c[0].toFixed(2)+" "+c[1].toFixed(2);}
    return d+(close?"Z":"");
  }
  function gRnd(seed){var s=seed>>>0;return function(){s=(s*1664525+1013904223)>>>0;return s/4294967296;};}
  function bez(c,pts,close){
    var p=pts.slice(),i;
    if(close){p.unshift(pts[pts.length-1]);p.push(pts[0],pts[1]);}
    else{p.unshift(pts[0]);p.push(pts[pts.length-1]);}
    c.moveTo(p[1][0],p[1][1]);
    for(i=1;i<p.length-2;i++){
      var a=p[i-1],b=p[i],d=p[i+1],e=p[i+2];
      c.bezierCurveTo(b[0]+(d[0]-a[0])/6,b[1]+(d[1]-a[1])/6,
                      d[0]-(e[0]-b[0])/6,d[1]-(e[1]-b[1])/6,d[0],d[1]);
    }
    if(close)c.closePath();
  }
  function dab(x,y,rx,ry,ang,col,a){
    emit(function(c){
      c.save();c.translate(x,y);if(ang)c.rotate(ang);c.scale(1,ry/rx);
      c.fillStyle=rgbStr(col,a);c.beginPath();c.arc(0,0,rx,0,6.2832);c.fill();c.restore();
    });
  }
  /* a loaded gouache stroke */
  function brush(pts,col,o){
    o=o||{};
    var C=typeof col==="string"?hexToRgb(col):col;
    var r=gRnd(o.seed||((pts[0][0]*13+pts[0][1]*7)|0)+1);
    var w=o.w||14,a=o.a==null?1:o.a;
    var path=[],i,k;
    for(i=0;i<pts.length-1;i++){
      var A=pts[i],B=pts[i+1];
      var d=Math.hypot(B[0]-A[0],B[1]-A[1]),n=Math.max(2,Math.round(d/(w*.16)));
      for(k=0;k<n;k++){var u=k/n;path.push([A[0]+(B[0]-A[0])*u,A[1]+(B[1]-A[1])*u]);}
    }
    path.push(pts[pts.length-1]);
    var L=path.length;
    for(i=0;i<L;i++){
      var t=i/(L-1);
      var pr=o.flat?1:.26+Math.sin(Math.pow(t,o.bias||.85)*Math.PI)*.74;
      var p=path[i];
      var ang=i<L-1?Math.atan2(path[i+1][1]-p[1],path[i+1][0]-p[0]):0;
      if(o.dry&&r()<o.dry)continue;
      var cc=r()<.3?lite(C,.05+r()*.1):(r()<.22?dark(C,.06):C);
      dab(p[0]+(r()-.5)*w*.07,p[1]+(r()-.5)*w*.07,
          w*pr*.5*(.93+r()*.14),w*pr*.5*(o.squash==null?.84:o.squash),ang,cc,a);
    }
    if(o.chalk!==false)for(i=0;i<L;i+=3){
      if(r()>.36)continue;var q=path[i];
      dab(q[0]+(r()-.5)*w*.9,q[1]+(r()-.5)*w*.7,
        .5+r()*1.4,.5+r()*1.2,0,lite(C,.24+r()*.2),a*(.35+r()*.4));
    }
  }
  /* a scrubbed field of gouache. the dabs of one brush pass are emitted as a
     single op, so the paint can be laid far more densely than the recorder
     would otherwise stand. */
